# 系统设计模式

> War3 地图开发中常见的系统设计模式。

## 1. 单位索引系统（Unit Indexer）

**目的**：为每个单位分配唯一索引，便于用数组存储单位数据（替代哈希表，性能更好）。

```lua
-- Lua 版
local unitIndex = 0
local unitList = {}
local recycleStack = {}

local function IndexUnit(u)
    local index
    if #recycleStack > 0 then
        index = table.remove(recycleStack)
    else
        unitIndex = unitIndex + 1
        index = unitIndex
    end
    unitList[index] = u
    SetUnitUserData(u, index)
    return index
end

local function DeindexUnit(u)
    local index = GetUnitUserData(u)
    if index > 0 then
        unitList[index] = nil
        table.insert(recycleStack, index)
        SetUnitUserData(u, 0)
    end
end
```

**应用**：存储单位自定义数据、技能状态、buff 等。

## 2. 计时器系统（Timer System）

**目的**：用单个计时器驱动多个周期任务，减少计时器数量。

```lua
local timers = {}
local PERIOD = 0.03125

local function RegisterPeriodic(callback, interval)
    table.insert(timers, {
        callback = callback,
        interval = interval,
        elapsed = 0
    })
end

local masterTimer = CreateTimer()
TimerStart(masterTimer, PERIOD, true, function()
    for i = #timers, 1, -1 do
        local t = timers[i]
        t.elapsed = t.elapsed + PERIOD
        if t.elapsed >= t.interval then
            t.elapsed = t.elapsed - t.interval
            t.callback()
        end
    end
end)
```

**应用**：投射物、持续伤害、周期性效果。

## 4. 伤害系统（Damage System）

**目的**：统一处理伤害事件，支持修改、反射、吸收等。

```lua
local damageHandlers = {}

local function RegisterDamageHandler(handler)
    table.insert(damageHandlers, handler)
end

local t = CreateTrigger()
TriggerRegisterAnyUnitEventBJ(t, EVENT_PLAYER_UNIT_DAMAGED)
TriggerAddAction(t, function()
    local target = GetTriggerUnit()
    local source = GetEventDamageSource()
    local damage = GetEventDamage()

    for _, handler in ipairs(damageHandlers) do
        damage = handler(target, source, damage)
    end

end)
```

## 5. 技能系统（Spell System）

**目的**：用假技能 + 触发器实现自定义技能。

```lua
local spellEffects = {}

local function RegisterSpell(abilId, onCast)
    spellEffects[abilId] = onCast
end

local t = CreateTrigger()
TriggerRegisterAnyUnitEventBJ(t, EVENT_PLAYER_UNIT_SPELL_EFFECT)
TriggerAddAction(t, function()
    local abilId = GetSpellAbilityId()
    local handler = spellEffects[abilId]
    if handler then
        handler(GetTriggerUnit(), GetSpellTargetUnit(), GetSpellTargetX(), GetSpellTargetY())
    end
end)
```

## 10. 事件总线（Event Bus）

**目的**：解耦系统间的通信。

```lua
local listeners = {}

local function On(event, callback)
    listeners[event] = listeners[event] or {}
    table.insert(listeners[event], callback)
end

local function Emit(event, ...)
    local handlers = listeners[event]
    if handlers then
        for _, h in ipairs(handlers) do
            h(...)
        end
    end
end
```

## 11. 状态机（State Machine）

**目的**：管理复杂的状态转换（如 Boss AI 阶段）。

```lua
local StateMachine = {}
StateMachine.__index = StateMachine

function StateMachine.new(states, initial)
    return setmetatable({
        states = states,
        current = initial
    }, StateMachine)
end

function StateMachine:transition(newState)
    if self.states[self.current].exit then
        self.states[self.current].exit()
    end
    self.current = newState
    if self.states[newState].enter then
        self.states[newState].enter()
    end
end

function StateMachine:update()
    local state = self.states[self.current]
    if state.update then state.update() end
end
```

## 12. 性能优化模式

### 12.1 对象池

```lua
local pool = {}

local function Acquire()
    if #pool > 0 then
        return table.remove(pool)
    end
    return CreateTimer()  -- 或其他对象
end

local function Release(obj)
    table.insert(pool, obj)
end
```

### 12.2 缓存

```lua
local cache = {}
local function GetCached(key, factory)
    if cache[key] == nil then
        cache[key] = factory()
    end
    return cache[key]
end
```

### 12.3 批量操作
